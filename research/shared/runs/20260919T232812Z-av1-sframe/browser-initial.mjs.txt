import {chromium} from 'playwright';import {createServer} from 'node:http';import {readFile,writeFile} from 'node:fs/promises';import assert from 'node:assert/strict';const out=process.argv[2],input=JSON.parse(await readFile(out+'/input.json','utf8'));const server=createServer((q,r)=>{r.setHeader('Content-Type','text/html');r.end('<body>');});await new Promise(r=>server.listen(0,'127.0.0.1',r));let browser;const result={scope:input.scope,cases:[]};try{browser=await chromium.launch({channel:'chrome',headless:true});result.browser=browser.version();const page=await browser.newPage();await page.goto('http://127.0.0.1:'+server.address().port);for(const target of [1,2])for(const wrong of [false,true]){const row=await page.evaluate(async({input,target,wrong})=>{const jobs=[],errors=[];let closed=0;const start=performance.now(),d=new VideoDecoder({output:f=>jobs.push((async()=>{try{const raw=new Uint8Array(f.allocationSize()),l=await f.copyTo(raw),b=new Uint8Array(23040);if(f.format!=='I420')throw Error(f.format);for(let y=0;y<96;y++)b.set(raw.subarray(l[0].offset+y*l[0].stride,l[0].offset+y*l[0].stride+160),y*160);for(let c=1;c<3;c++)for(let y=0;y<48;y++)b.set(raw.subarray(l[c].offset+y*l[c].stride,l[c].offset+y*l[c].stride+80),15360+(c-1)*3840+y*80);return {timestamp:f.timestamp,hash:[...new Uint8Array(await crypto.subtle.digest('SHA-256',b))].map(x=>x.toString(16).padStart(2,'0')).join('')};}finally{f.close();closed++;}})()),error:e=>errors.push(String(e))});try{d.configure({codec:'av01.0.00M.08',codedWidth:160,codedHeight:96,hardwareAcceleration:'prefer-software'});for(let i=0;i<12;i++){const packet=input.packets[i<4?(wrong?3:0):target][i];d.decode(new EncodedVideoChunk({type:i?'delta':'key',timestamp:Math.round(i*1e6/24),data:new Uint8Array(packet)}));}await d.flush();const frames=await Promise.all(jobs);d.reset();d.configure({codec:'av01.0.00M.08',codedWidth:160,codedHeight:96});let coldRejected=false;try{d.decode(new EncodedVideoChunk({type:'delta',timestamp:0,data:new Uint8Array(input.packets[target][4])}));}catch(e){coldRejected=e.name==='DataError';}d.close();return {target,wrong,ms:performance.now()-start,frames:frames.length,closed,exact:frames.length===12&&frames.every((f,i)=>f.hash===input.hashes[target][i]&&f.timestamp===Math.round(i*1e6/24)),errors,coldRejected};}finally{if(d.state!=='closed')d.close();}},{input,target,wrong});assert.equal(row.exact,!wrong);assert.ok(row.coldRejected&&row.closed===12&&row.errors.length===0);result.cases.push(row);}result.passed=true;result.analysis='Both prepared S-frame and ordinary inter continuation are exact with the identical shared prefix; both wrong histories fail full-picture identity. Thus no additional switch opportunity in this fixed-config shared-history profile. Neither is accepted as a cold-start delta. No general S-frame rejection or independent-history switch claim.';}catch(e){result.error=String(e.stack);process.exitCode=1;}finally{await browser?.close();await new Promise(r=>server.close(r));await writeFile(out+'/browser-result.json',JSON.stringify(result,null,2)+'\n');console.log(JSON.stringify(result));}
